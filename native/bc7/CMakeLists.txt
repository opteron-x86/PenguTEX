cmake_minimum_required(VERSION 3.20)
project(PenguTEXBc7 LANGUAGES CXX)
find_program(ISPC_EXECUTABLE ispc REQUIRED)
execute_process(COMMAND "${ISPC_EXECUTABLE}" --version OUTPUT_VARIABLE ISPC_VERSION)
if(NOT ISPC_VERSION MATCHES "1\\.32\\.0")
  message(FATAL_ERROR "Use pinned ISPC 1.32.0 for reproducible encoder builds")
endif()
set(EXT o)
if(WIN32)
  set(EXT obj)
endif()
set(OBJECTS "${CMAKE_CURRENT_BINARY_DIR}/bc7e.${EXT}"
  "${CMAKE_CURRENT_BINARY_DIR}/bc7e_sse2.${EXT}" "${CMAKE_CURRENT_BINARY_DIR}/bc7e_avx2.${EXT}")
set(PIC)
if(NOT WIN32)
  set(PIC --pic)
endif()
add_custom_command(OUTPUT ${OBJECTS} "${CMAKE_CURRENT_BINARY_DIR}/bc7e.h"
  COMMAND "${ISPC_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/kernel.ispc" -O2 ${PIC}
    --arch=x86-64 --target=sse2-i32x4,avx2-i32x8 --opt=disable-assertions
    -o "${CMAKE_CURRENT_BINARY_DIR}/bc7e.${EXT}" -h "${CMAKE_CURRENT_BINARY_DIR}/bc7e.h"
  DEPENDS kernel.ispc vendor/bc7e.ispc VERBATIM)
add_library(pengutex_bc7 SHARED encoder.cpp ${OBJECTS} "${CMAKE_CURRENT_BINARY_DIR}/bc7e.h")
target_compile_features(pengutex_bc7 PRIVATE cxx_std_17)
target_include_directories(pengutex_bc7 PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
set_target_properties(pengutex_bc7 PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(MSVC)
  set_property(TARGET pengutex_bc7 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(pengutex_bc7 PRIVATE -Wall -Wextra)
  # CI treats warnings as errors; source builds with a newer compiler should still succeed.
  if(PENGUTEX_WARNINGS_AS_ERRORS)
    target_compile_options(pengutex_bc7 PRIVATE -Werror)
  endif()
endif()
install(TARGETS pengutex_bc7 RUNTIME DESTINATION . LIBRARY DESTINATION .)
include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(bc7_native_tests tests.cpp)
  target_compile_features(bc7_native_tests PRIVATE cxx_std_17)
  target_link_libraries(bc7_native_tests PRIVATE pengutex_bc7 Threads::Threads)
  add_test(NAME bc7_native COMMAND bc7_native_tests)
endif()
